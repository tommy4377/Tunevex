Write-Host "Configuring GPU interrupt affinity..." -ForegroundColor Yellow

$vendors = @('VEN_10DE', 'VEN_1002', 'VEN_8086')
$devices = Get-PnpDevice -Class Display -Status OK -EA SilentlyContinue | Where-Object {
    foreach ($vendor in $vendors) {
        if ($_.InstanceId -match $vendor) { return $true }
    }
    return $false
}

foreach ($dev in $devices) {
    $affinityPath = "HKLM:\SYSTEM\CurrentControlSet\Enum\$($dev.InstanceId)\Device Parameters\Interrupt Management\Affinity Policy"
    
    if (-not (Test-Path $affinityPath)) { 
        New-Item -Path $affinityPath -Force | Out-Null 
    }
    
    # DevicePolicy = 4: IrqPolicySpecifiedProcessors
    # AssignmentSetOverride: CPU affinity mask (0x0F = cores 0-3)
    Set-ItemProperty -Path $affinityPath -Name 'DevicePolicy' -Value 4 -Type DWord -Force
    Set-ItemProperty -Path $affinityPath -Name 'AssignmentSetOverride' -Value 0x0F -Type DWord -Force
    
    Write-Host "  Configured: $($dev.FriendlyName)" -ForegroundColor Green
}

#[command]
pub async fn create_restore_point(description: String) -> Result<String, String> {
    // Check if System Protection is enabled first
    let check_script = r#"
$os = Get-WmiObject Win32_OperatingSystem
$systemDrive = $os.SystemDrive
$rp = Get-ComputerRestorePoint -EA 0 | Select-Object -First 1

# Check if protection is enabled
$protected = (Get-ItemProperty -Path "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SystemRestore" -Name "RPSessionInterval" -EA 0) -ne $null

if (-not $protected) {
    Write-Output "DISABLED"
    exit 1
}

# Check if enough space (need at least 300MB)
$freeSpace = (Get-Volume -DriveLetter $systemDrive.TrimEnd(':') | Select-Object -ExpandProperty SizeRemaining) / 1GB
if ($freeSpace -lt 0.3) {
    Write-Output "NO_SPACE"
    exit 1
}

# Check 24-hour limit
if ($rp -and ((Get-Date) - $rp.CreationTime).TotalHours -lt 24) {
    Write-Output "TOO_SOON"
    exit 1
}

Write-Output "OK"
exit 0
"#;

    let check_output = Command::new("powershell")
        .args(&["-NoProfile", "-Command", check_script])
        .output()
        .map_err(|e| e.to_string())?;

    let check_result = String::from_utf8_lossy(&check_output.stdout).trim().to_string();
    
    match check_result.as_str() {
        "DISABLED" => return Err("System Protection is disabled. Enable it in System Properties -> System Protection.".to_string()),
        "NO_SPACE" => return Err("Insufficient disk space for restore point (need 300MB free).".to_string()),
        "TOO_SOON" => return Err("A restore point was already created in the last 24 hours. Windows limits frequency.".to_string()),
        _ => {} // OK, proceed
    }

    // Create restore point
    let output = Command::new("powershell")
        .args(&[
            "-NoProfile",
            "-Command",
            &format!(
                "Checkpoint-Computer -Description '{}' -RestorePointType 'MODIFY_SETTINGS'",
                description
            ),
        ])
        .output()
        .map_err(|e| e.to_string())?;

    if output.status.success() {
        Ok("Restore point created successfully.".to_string())
    } else {
        Err(String::from_utf8_lossy(&output.stderr).to_string())
    }
}

// uxtheme_hook/src/lib.rs
use windows::Win32::Foundation::*;
use windows::Win32::System::SystemServices::*;
use detour::static_detour;

static_detour! {
    static VerifyThemeSignature: unsafe extern "system" fn(*const u16) -> BOOL;
}

// Hook function that always returns TRUE (signature valid)
unsafe extern "system" fn verify_theme_signature_hook(_path: *const u16) -> BOOL {
    TRUE  // Always allow themes
}

#[no_mangle]
#[allow(non_snake_case)]
extern "system" fn DllMain(
    _dll_module: HINSTANCE,
    call_reason: u32,
    _: *mut ()
) -> BOOL {
    match call_reason {
        DLL_PROCESS_ATTACH => {
            // Initialize hooks when DLL is loaded
            unsafe {
                let uxtheme = LoadLibraryW(w!("uxtheme.dll")).unwrap();
                let target_fn = GetProcAddress(uxtheme, s!("VerifyThemeSignature"));
                
                if let Some(target) = target_fn {
                    VerifyThemeSignature.initialize(
                        std::mem::transmute(target), 
                        verify_theme_signature_hook
                    ).unwrap();
                    VerifyThemeSignature.enable().unwrap();
                }
            }
        }
        DLL_PROCESS_DETACH => {
            unsafe {
                let _ = VerifyThemeSignature.disable();
            }
        }
        _ => {}
    }
    TRUE
}

// Add to boot.rs
pub fn toggle_boot_item(id: &str, enable: bool) -> Result<(), String> {
    if id.starts_with("IFEO:") {
        let exe_name = id.strip_prefix("IFEO:").ok_or("Invalid IFEO ID")?;
        let path = format!(r"SOFTWARE\Microsoft\Windows NT\CurrentVersion\Image File Execution Options\{}", exe_name);
        
        let key = RegKey::predef(HKEY_LOCAL_MACHINE)
            .open_subkey_with_flags(&path, KEY_ALL_ACCESS)
            .map_err(|e| e.to_string())?;
        
        if enable {
            return Err("Cannot re-enable IFEO hijacks (too dangerous)".to_string());
        } else {
            // BACKUP FIRST
            let debugger: String = key.get_value("Debugger").map_err(|e| e.to_string())?;
            let backup_path = format!(r"Software\TommyTweaker\IFEOBackup\{}", exe_name);
            let (backup_key, _) = RegKey::predef(HKEY_CURRENT_USER)
                .create_subkey(&backup_path)
                .map_err(|e| e.to_string())?;
            backup_key.set_value("Debugger", &debugger)?;
            
            // DELETE
            key.delete_value("Debugger").map_err(|e| e.to_string())?;
        }
    } else if id.starts_with("APPINIT:") {
        // Similar logic for AppInit_DLLs - clear the value
        return Err("AppInit_DLL removal requires manual registry edit".to_string());
    }
    
    Ok(())
}

pub fn assess_safety(path: &str, publisher: Option<&str>) -> SafetyRating {
    let path_lower = path.to_lowercase();
    let pub_lower = publisher.unwrap_or("").to_lowercase();
    
    // Safe publishers/locations
    if pub_lower.contains("microsoft") || pub_lower.contains("google")
        || path_lower.contains("windows\\system32")
        || path_lower.contains("windows\\syswow64")
    {
        return SafetyRating::Safe;
    }
    
    // Careful (known apps)
    if path_lower.contains("steam") || path_lower.contains("discord")
        || path_lower.contains("spotify") || pub_lower.contains("valve")
        || pub_lower.contains("nvidia") || pub_lower.contains("intel")
        || pub_lower.contains("adobe") || pub_lower.contains("epic games")
    {
        return SafetyRating::Careful;
    }
    
    // 🔴 DANGEROUS INDICATORS (malware patterns)
    if path_lower.contains("temp") 
        || path_lower.contains("appdata\\local\\temp")
        || path_lower.contains("public\\")  // C:\Users\Public\ is suspicious
        || path_lower.contains("programdata\\")  // ProgramData for non-MS apps is risky
        || path_lower.contains(".tmp")
        || path_lower.contains("downloads\\")
        || path_lower.ends_with(".vbs")  // Script files
        || path_lower.ends_with(".bat")
        || path_lower.ends_with(".cmd")
        || path_lower.ends_with(".ps1")
    {
        return SafetyRating::Dangerous;
    }
    
    // If no publisher and not in safe locations
    if publisher.is_none() {
        return SafetyRating::Unknown;
    }
    
    SafetyRating::Unknown
}

// Add to browser.rs

fn scan_chrome_extensions() -> Vec<StartupItem> {
    let mut items = Vec::new();
    
    // Chrome Extensions folder
    let local_appdata = std::env::var("LOCALAPPDATA").ok()?;
    let chrome_path = PathBuf::from(local_appdata)
        .join(r"Google\Chrome\User Data\Default\Extensions");
    
    if !chrome_path.exists() {
        return items;
    }
    
    // Scan extension IDs
    for entry in std::fs::read_dir(&chrome_path).ok()?.flatten() {
        let ext_id = entry.file_name().to_string_lossy().to_string();
        
        // Read manifest.json for details
        let manifest_path = entry.path().join("manifest.json");
        if let Ok(manifest_data) = std::fs::read_to_string(&manifest_path) {
            if let Ok(json) = serde_json::from_str::<Value>(&manifest_data) {
                let name = json["name"].as_str().unwrap_or(&ext_id).to_string();
                
                items.push(StartupItem {
                    id: format!("CHROME_EXT:{}", ext_id),
                    name,
                    category: "Browser".to_string(),
                    subcategory: "Chrome Extension".to_string(),
                    location: chrome_path.to_string_lossy().to_string(),
                    command: ext_id,
                    enabled: true,
                    publisher: json["author"].as_str().map(String::from),
                    description: json["description"].as_str().map(String::from),
                    source: AutostartSource::FileSystem,
                    safety_rating: SafetyRating::Unknown,
                    file_exists: true,
                });
            }
        }
    }
    
    items
}

        let path = format!(r"SOFTWARE\Microsoft\Windows NT\CurrentVersion\Image File Execution Options\{}", exe_name);

            .open_subkey_with_flags(&path, KEY_ALL_ACCESS)
            .map_err(|e| e.to_string())?;
        
        if enable {
            return Err("Cannot re-enable IFEO hijacks (too dangerous)".to_string());
        } else {
            // BACKUP FIRST
            let debugger: String = key.get_value("Debugger").map_err(|e| e.to_string())?;
            let backup_path = format!(r"Software\TommyTweaker\IFEOBackup\{}", exe_name);
            let (backup_key, _) = RegKey::predef(HKEY_CURRENT_USER)
                .create_subkey(&backup_path)
                .map_err(|e| e.to_string())?;

pub fn assess_safety(path: &str, publisher: Option<&str>) -> SafetyRating {
    let path_lower = path.to_lowercase();
    let pub_lower = publisher.unwrap_or("").to_lowercase();
    
    // Safe publishers/locations
    if pub_lower.contains("microsoft") || pub_lower.contains("google")
        || path_lower.contains("windows\\system32")
        || path_lower.contains("windows\\syswow64")
    {
        return SafetyRating::Safe;
    }
    
    // Careful (known apps)
    if path_lower.contains("steam") || path_lower.contains("discord")
        || path_lower.contains("spotify") || pub_lower.contains("valve")
        || pub_lower.contains("nvidia") || pub_lower.contains("intel")
        || pub_lower.contains("adobe") || pub_lower.contains("epic games")
    {
        return SafetyRating::Careful;
    }
    
    // 🔴 DANGEROUS INDICATORS (malware patterns)
    if path_lower.contains("temp") 
        || path_lower.contains("appdata\\local\\temp")
        || path_lower.contains("public\\")  // C:\Users\Public\ is suspicious
        || path_lower.contains("programdata\\")  // ProgramData for non-MS apps is risky
        || path_lower.contains(".tmp")
        || path_lower.contains("downloads\\")
        || path_lower.ends_with(".vbs")  // Script files
        || path_lower.ends_with(".bat")
        || path_lower.ends_with(".cmd")
        || path_lower.ends_with(".ps1")
    {
        return SafetyRating::Dangerous;
    }
    
    // If no publisher and not in safe locations
    if publisher.is_none() {
        return SafetyRating::Unknown;
    }
    
    SafetyRating::Unknown
}

// Add to browser.rs

fn scan_chrome_extensions() -> Vec<StartupItem> {
    let mut items = Vec::new();
    
    // Chrome Extensions folder
    let local_appdata = std::env::var("LOCALAPPDATA").ok()?;
    let chrome_path = PathBuf::from(local_appdata)
        .join(r"Google\Chrome\User Data\Default\Extensions");
    
    if !chrome_path.exists() {
        return items;
    }
    
    // Scan extension IDs
    for entry in std::fs::read_dir(&chrome_path).ok()?.flatten() {
        let ext_id = entry.file_name().to_string_lossy().to_string();
        
        // Read manifest.json for details
        let manifest_path = entry.path().join("manifest.json");
        if let Ok(manifest_data) = std::fs::read_to_string(&manifest_path) {
            if let Ok(json) = serde_json::from_str::<Value>(&manifest_data) {
                let name = json["name"].as_str().unwrap_or(&ext_id).to_string();
                
                items.push(StartupItem {
                    id: format!("CHROME_EXT:{}", ext_id),
                    name,
                    category: "Browser".to_string(),
                    subcategory: "Chrome Extension".to_string(),
                    location: chrome_path.to_string_lossy().to_string(),
                    command: ext_id,
                    enabled: true,
                    publisher: json["author"].as_str().map(String::from),
                    description: json["description"].as_str().map(String::from),
                    source: AutostartSource::FileSystem,
                    safety_rating: SafetyRating::Unknown,
                    file_exists: true,
                });
            }
        }
    }
    
    items
}

pub fn is_compressed(path: &Path) -> bool {
    unsafe {
        let path_str = HSTRING::from(path.as_os_str());
        let handle = CreateFileW(
            PCWSTR::from_raw(path_str.as_ptr()),
            FILE_READ_DATA.0,
            FILE_SHARE_READ | FILE_SHARE_WRITE,
            None,
            OPEN_EXISTING,
            FILE_FLAG_BACKUP_SEMANTICS,
            None,
        );
        
        if handle.is_err() || handle.as_ref().unwrap() == &INVALID_HANDLE_VALUE {
            return false;
        }
        
        let handle = handle.unwrap();
        let _handle_guard = HandleGuard(handle);
        
        const FSCTL_GET_EXTERNAL_BACKING: u32 = 0x90310;
        
        let mut output_buffer: [u8; 256] = [0; 256];
        let mut bytes_returned: u32 = 0;
        
        let result = DeviceIoControl(
            handle,
            FSCTL_GET_EXTERNAL_BACKING,
            None,
            0,
            Some(output_buffer.as_mut_ptr() as *mut _),
            output_buffer.len() as u32,
            Some(&mut bytes_returned),
            None,
        );
        
        result.is_ok() && bytes_returned > 0
    }
}

#[command]
pub async fn search_packages(query: String) -> Result<Vec<WingetPackage>, String> {
    let output = Command::new("powershell")
        .args(&[
            "-NoProfile",
            "-Command",
            &format!("winget search \"{}\" --accept-source-agreements", query),
        ])
        .output()
        .map_err(|e| e.to_string())?;

    if !output.status.success() {
        return Err("Search failed".to_string());
    }

    let stdout = String::from_utf8_lossy(&output.stdout);
    let mut packages = Vec::new();

    // Parse search results (simplified)
    for line in stdout.lines().skip(2) {
        if line.trim().is_empty() {
            continue;
        }

        let parts: Vec<&str> = line.split_whitespace().collect();
        if parts.len() >= 2 {
            packages.push(WingetPackage {
                name: parts.get(0).unwrap_or(&"").to_string(),
                id: parts.get(1).unwrap_or(&"").to_string(),
                description: "".to_string(),
                category: "Search Result".to_string(),
                installed: false,
            });
        }
    }

    Ok(packages)
}
